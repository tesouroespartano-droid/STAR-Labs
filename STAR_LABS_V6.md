                                                                                                                                                                                                                                                                                                                                                                                                                                                                                           test -s "$APK"
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                           stat "$APK"
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                           unzip -t "$APK"
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                           apksigner verify --verbose "$APK"

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                           e registre tamanho e SHA-256.

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                           Se o arquivo tiver 0 bytes em qualquer etapa, o build deve falhar imediatamente.
